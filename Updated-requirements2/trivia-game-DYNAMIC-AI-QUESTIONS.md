```javascript
class AdaptiveDifficultyEngine {
  constructor() {
    this.playerProfile = {
      questionsAnswered: 0,
      correctAnswers: 0,
      currentStreak: 0,
      longestStreak: 0,
      avgResponseTime: 0,
      domainKnowledge: {
        aiAgents: { correct: 0, asked: 0 },
        productionSystems: { correct: 0, asked: 0 },
        distributedSystems: { correct: 0, asked: 0 },
        aiProduction: { correct: 0, asked: 0 }
      },
      failureTopics: [], // Track what they got wrong
      speedPattern: 'normal', // slow, normal, fast, rushing
      confidence: 0.5 // 0-1 scale
    };
  }

  calculateCurrentDifficulty() {
    const accuracy = this.playerProfile.correctAnswers / 
                     this.playerProfile.questionsAnswered;
    
    // Base difficulty on accuracy
    let difficulty = 0.5; // Start medium
    
    if (accuracy >= 0.9) difficulty = 0.9; // Very hard
    else if (accuracy >= 0.75) difficulty = 0.75; // Hard
    else if (accuracy >= 0.6) difficulty = 0.6; // Medium-hard
    else if (accuracy >= 0.4) difficulty = 0.5; // Medium
    else if (accuracy >= 0.2) difficulty = 0.35; // Medium-easy
    else difficulty = 0.2; // Easy
    
    // Adjust for streak
    if (this.playerProfile.currentStreak >= 4) {
      difficulty += 0.1; // Make it harder
    }
    
    // Adjust for speed (rushing might mean guessing)
    if (this.playerProfile.speedPattern === 'rushing') {
      difficulty += 0.05; // Punish guessing with harder questions
    }
    
    return Math.min(0.95, Math.max(0.1, difficulty));
  }

  shouldGenerateRecoveryQuestion() {
    // Generate funny "recovery" question if:
    // 1. Just got a question wrong
    // 2. Streak was broken
    // 3. Confidence is dropping
    const justFailed = this.playerProfile.lastAnswerCorrect === false;
    const streakBroken = this.playerProfile.currentStreak === 0 && 
                         this.playerProfile.lastStreak >= 3;
    
    return justFailed && (streakBroken || this.playerProfile.confidence < 0.3);
  }

  getNextQuestionParams() {
    const difficulty = this.calculateCurrentDifficulty();
    const needsRecovery = this.shouldGenerateRecoveryQuestion();
    
    // Select topic based on domain knowledge
    const weakestDomain = this.findWeakestDomain();
    const strongestDomain = this.findStrongestDomain();
    
    return {
      difficulty,
      needsRecovery,
      targetDomain: difficulty > 0.7 ? strongestDomain : weakestDomain,
      avoidTopics: this.playerProfile.failureTopics.slice(-2), // Don't repeat recent failures
      playerContext: {
        streak: this.playerProfile.currentStreak,
        accuracy: this.playerProfile.correctAnswers / this.playerProfile.questionsAnswered,
        questionsRemaining: 12 - this.playerProfile.questionsAnswered
      }
    };
  }
}

```javascript
// services/questionGenerator.js

import { CLAUDE_CONFIG } from '../config/claude.js';

export class QuestionGeneratorService {
  constructor() {
    this.cache = new Map(); // Cache recent questions to avoid repeats
    this.generatedCount = 0;
  }

  async generateQuestion(params) {
    const { needsRecovery, isSurprise } = params;
    
    let prompt;
    if (needsRecovery) {
      prompt = generateRecoveryQuestionPrompt(
        params.playerContext, 
        params.lastFailedTopic
      );
    } else if (isSurprise) {
      prompt = generateSurpriseQuestionPrompt();
    } else {
      prompt = generateStandardQuestionPrompt(params);
    }

    try {
      const response = await fetch(CLAUDE_CONFIG.apiEndpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'anthropic-api-key': CLAUDE_CONFIG.apiKey,
          'anthropic-version': '2023-06-01'
        },
        body: JSON.stringify({
          model: CLAUDE_CONFIG.model,
          max_tokens: CLAUDE_CONFIG.maxTokens,
          messages: [{
            role: 'user',
            content: prompt
          }]
        })
      });

      const data = await response.json();
      const questionText = data.content[0].text;
      
      // Parse JSON response
      const question = this.parseQuestionJSON(questionText);
      
      // Validate question quality
      if (!this.validateQuestion(question)) {
        console.warn('Generated question failed validation, retrying...');
        return await this.generateQuestion(params); // Retry once
      }
      
      // Shuffle answers
      question.answers = this.shuffleAnswers(question.answers);
      
      // Add metadata
      question.id = `claude_${Date.now()}_${this.generatedCount++}`;
      question.generatedAt = new Date().toISOString();
      
      // Cache to avoid near-duplicates
      this.cacheQuestion(question);
      
      return question;
      
    } catch (error) {
      console.error('Question generation failed:', error);
      // Fallback to a safe default question
      return this.getFallbackQuestion(params);
    }
  }

  parseQuestionJSON(text) {
    // Claude sometimes wraps JSON in markdown code blocks
    let cleanText = text.trim();
    
    // Remove markdown code blocks if present
    cleanText = cleanText.replace(/```json\n?/g, '');
    cleanText = cleanText.replace(/```\n?/g, '');
    cleanText = cleanText.trim();
    
    try {
      return JSON.parse(cleanText);
    } catch (e) {
      console.error('Failed to parse question JSON:', cleanText);
      throw new Error('Invalid question JSON from Claude');
    }
  }

  validateQuestion(question) {
    // Ensure question meets quality standards
    const checks = [
      // Has all required fields
      question.question && question.question.length > 20,
      Array.isArray(question.answers) && question.answers.length === 4,
      question.explanation && question.explanation.length > 10,
      
      // Exactly one correct answer
      question.answers.filter(a => a.correct).length === 1,
      
      // All answers have reasonable length
      question.answers.every(a => a.text.length > 10 && a.text.length < 200),
      
      // No obviously silly answers (heuristic)
      !question.answers.some(a => 
        a.text.toLowerCase().includes('magic') ||
        a.text.toLowerCase().includes('wizard') ||
        a.text.toLowerCase().includes('unicorn')
      ),
      
      // Answers are diverse (not too similar)
      this.checkAnswerDiversity(question.answers)
    ];
    
    return checks.every(check => check === true);
  }

  checkAnswerDiversity(answers) {
    // Check that answers aren't too similar
    const texts = answers.map(a => a.text.toLowerCase());
    
    for (let i = 0; i < texts.length; i++) {
      for (let j = i + 1; j < texts.length; j++) {
        const similarity = this.calculateSimilarity(texts[i], texts[j]);
        if (similarity > 0.8) return false; // Too similar
      }
    }
    
    return true;
  }

  calculateSimilarity(str1, str2) {
    // Simple word overlap similarity
    const words1 = new Set(str1.split(' '));
    const words2 = new Set(str2.split(' '));
    const intersection = new Set([...words1].filter(x => words2.has(x)));
    const union = new Set([...words1, ...words2]);
    return intersection.size / union.size;
  }

  shuffleAnswers(answers) {
    // Shuffle while preserving correct answer reference
    const shuffled = [...answers].sort(() => Math.random() - 0.5);
    return shuffled.map((answer, index) => ({
      ...answer,
      id: ['a', 'b', 'c', 'd'][index]
    }));
  }

  cacheQuestion(question) {
    const key = this.generateQuestionHash(question);
    this.cache.set(key, {
      question,
      timestamp: Date.now()
    });
    
    // Clear old cache entries (older than 1 hour)
    const oneHourAgo = Date.now() - 3600000;
    for (const [key, value] of this.cache.entries()) {
      if (value.timestamp < oneHourAgo) {
        this.cache.delete(key);
      }
    }
  }

  generateQuestionHash(question) {
    // Simple hash based on question text
    return question.question.toLowerCase().substring(0, 50);
  }

  getFallbackQuestion(params) {
    // Emergency fallback if Claude API fails
    return {
      id: `fallback_${Date.now()}`,
      question: "What does 'SRE' stand for?",
      answers: [
        { id: "a", text: "Site Reliability Engineering", correct: true },
        { id: "b", text: "System Resource Efficiency", correct: false },
        { id: "c", text: "Service Response Engineering", correct: false },
        { id: "d", text: "Server Reliability Execution", correct: false }
      ],
      explanation: "SRE (Site Reliability Engineering) is the practice of applying software engineering to operations.",
      difficulty: 0.3,
      domain: "production_systems",
      timer_seconds: 15,
      isFallback: true
    };
  }
}

## GAME FLOW WITH DYNAMIC QUESTIONS

### Question Selection Logic

```javascript
// Game controller
export class GameController {
  constructor(gameId, playerId) {
    this.gameId = gameId;
    this.playerId = playerId;
    this.difficultyEngine = new AdaptiveDifficultyEngine();
    this.questionGenerator = new QuestionGeneratorService();
    this.questionsAsked = [];
  }

  async getNextQuestion() {
    // Update player profile based on last answer
    if (this.lastAnswer) {
      this.difficultyEngine.updateFromAnswer(this.lastAnswer);
    }
    
    // Determine next question parameters
    const params = this.difficultyEngine.getNextQuestionParams();
    
    // Every 3-4 questions, insert a surprise
    const shouldBeSurprise = this.shouldInsertSurprise();
    if (shouldBeSurprise) {
      params.isSurprise = true;
    }
    
    // Generate question via Claude
    const question = await this.questionGenerator.generateQuestion(params);
    
    // Store for tracking
    this.questionsAsked.push({
      questionId: question.id,
      difficulty: question.difficulty,
      domain: question.domain,
      timestamp: Date.now()
    });
    
    return question;
  }

  shouldInsertSurprise() {
    const questionNumber = this.questionsAsked.length + 1;
    const surprisePositions = [3, 7, 10]; // Fixed positions
    return surprisePositions.includes(questionNumber);
  }

  async handleAnswer(answer, timeRemaining) {
    const currentQuestion = this.questionsAsked[this.questionsAsked.length - 1];
    const isCorrect = answer.correct;
    
    // Update player profile
    this.difficultyEngine.recordAnswer({
      questionId: currentQuestion.questionId,
      correct: isCorrect,
      timeRemaining,
      difficulty: currentQuestion.difficulty,
      domain: currentQuestion.domain
    });
    
    // Calculate score
    const score = this.calculateScore(currentQuestion, timeRemaining, isCorrect);
    
    // Store for next question generation
    this.lastAnswer = {
      correct: isCorrect,
      domain: currentQuestion.domain,
      difficulty: currentQuestion.difficulty
    };
    
    return {
      correct: isCorrect,
      score,
      explanation: answer.explanation,
      shouldShowRecovery: !isCorrect && this.difficultyEngine.shouldGenerateRecoveryQuestion()
    };
  }
}

  // Upload to cloud storage or save locally
  const imageUrl = await uploadToStorage(buffer, `personality_${personality.id}_${Date.now()}.png`);
  
  return {
    imageUrl,
    buffer,
    dimensions: { width, height }
  };
}

## FALLBACK STRATEGIES

### If Claude API is Down

```javascript
class QuestionGeneratorService {
  async generateQuestion(params) {
    try {
      return await this.generateViaClaudeAPI(params);
    } catch (error) {
      console.error('Claude API failed, using fallback:', error);
      
      // Fallback 1: Use cached questions from previous games
      const cached = await this.getCachedQuestion(params);
      if (cached) return cached;
      
      // Fallback 2: Use pre-generated question bank
      return await this.getFromStaticBank(params);
    }
  }
  
  async getCachedQuestion(params) {
    // Pull from recent successful generations
    const recentQuestions = await db.query(`
      SELECT * FROM generated_questions
      WHERE difficulty BETWEEN $1 AND $2
        AND domain = $3
        AND generated_at > NOW() - INTERVAL '24 hours'
      ORDER BY RANDOM()
      LIMIT 1
    `, [params.difficulty - 0.1, params.difficulty + 0.1, params.targetDomain]);
    
    return recentQuestions.rows[0] || null;
  }
}
```
